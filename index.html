🛠️ (초기 1회용) Supabase SQL 테이블 설정 코드

처음 프로젝트를 만드셨다면 Supabase SQL Editor에서 아래 구문을 1회 실행해 주세요.

create table if not exists fixed_expenses (
  id text primary key, name text, amount bigint, base_amount bigint,
  is_variable boolean default false, day int, category text, type text,
  payment_method text, payment_detail text, memo text,
  duration_months int, start_year_month text, owner text default '남편'
);
alter table fixed_expenses add column if not exists owner text default '남편';
alter table fixed_expenses enable row level security;
drop policy if exists "anon all" on fixed_expenses;
create policy "anon all" on fixed_expenses for all using (true) with check (true);
alter publication supabase_realtime add table fixed_expenses;
